### E4：状态机三态流转（完整链路）

**验证目标**：验证 CONDITIONS_NOT_MET → IN_PROGRESS → COMPLETED 的完整流转

**实验操作**：

```powershell
# 创建独立实验节点 E4-001
docker exec emily-core python -c "
import urllib.request, json

def api(method, path, data=None):
    body = json.dumps(data).encode() if data else None
    req = urllib.request.Request(f'http://localhost:18080{path}',
        data=body, headers={'Content-Type': 'application/json'} if body else {}, method=method)
    if method in ('PATCH','DELETE'): req.get_method = lambda: method
    try:
        resp = urllib.request.urlopen(req)
        return json.loads(resp.read())
    except urllib.error.HTTPError as e:
        return {'error': e.code, 'detail': e.read().decode()}

# Step 1: 创建节点（无依赖、无成果）
r = api('POST', '/api/v1/project-nodes', {
    'project_id': 'exp-state-machine', 'node_id': 'E4-001',
    'node_name': '流转测试-土方工程', 'deadline': '2026-12-31T18:00:00+08:00',
    'creator_id': 'tester', 'owner_dept_id': '工程部', 'stage_id': 2
})
print(f'Step1 创建: status={r.get(\"data\",{}).get(\"status\")} (expected: CONDITIONS_NOT_MET)')

# Step 2: 添加成果（进度0%）
r2 = api('POST', '/api/v1/project-nodes/E4-001/deliverables', {
    'deliverable_name': '土方开挖验收单', 'target_amount': 10, 'unit': '份', 'is_required': True
})
print(f'Step2 添加成果: {r2.get(\"message\")}')

# Step 3: 查询当前状态
resp = urllib.request.urlopen('http://localhost:18080/api/v1/project-nodes/E4-001')
d = json.loads(resp.read())['data']
print(f'Step3 添加成果后: status={d[\"status\"]}, progress={d[\"progress\"]}')
# 无依赖 + 有成果(0%) → IN_PROGRESS
assert d['status'] == 'IN_PROGRESS', f'Expected IN_PROGRESS, got {d[\"status\"]}'
did = d['deliverables'][0]['deliverable_id']
print(f'  成果ID: {did}')

# Step 4: 逐步推进成果
for amt in [3, 6, 10]:
    r = api('PATCH', f'/api/v1/node-deliverables/{did}', {'current_amount': amt})
    resp = urllib.request.urlopen('http://localhost:18080/api/v1/project-nodes/E4-001')
    d = json.loads(resp.read())['data']
    print(f'Step4-{amt}: progress={d[\"progress\"]}, status={d[\"status\"]}')

# 最终验证
assert d['status'] == 'COMPLETED', f'Expected COMPLETED, got {d[\"status\"]}'
assert d['progress'] >= 100.0, f'Expected progress >= 100, got {d[\"progress\"]}'
print('[PASS] E4 状态机流转: CONDITIONS_NOT_MET -> IN_PROGRESS -> COMPLETED')
"
```

**判定标准**：最终 status=COMPLETED，progress=100.0。

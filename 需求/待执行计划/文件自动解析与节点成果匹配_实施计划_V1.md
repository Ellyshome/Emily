```python
async def update_summary(self, file_id: str, summary: str) -> None:
    """更新文件的 content_summary 和 summary_generated_at"""
    async with self.session_factory() as session:
        f = await session.get(File, file_id)
        if f and f.content_summary is None:
            f.content_summary = summary
            f.summary_generated_at = datetime.utcnow()
            await session.commit()

async def get_by_summary_null(self, limit: int = 50) -> list[File]:
    """取出 content_summary 为空的文件，按时间升序"""
    async with self.session_factory() as session:
        result = await session.execute(
            select(File)
            .where(File.content_summary.is_(None))
            .order_by(File.created_at.asc())
            .limit(limit)
        )
        return result.scalars().all()
```

```python
import asyncio
import logging
from dataclasses import dataclass
from pathlib import Path

logger = logging.getLogger(__name__)

MAX_PAGES = 10
MAX_FILE_SIZE_MB = 50


@dataclass
class ParseResult:
    summary: str          # 文件摘要
    file_type: str        # 证照/合同/图纸/公文/报告/其他
    key_info: dict        # {证号, 日期, 甲方, 乙方, ...}
    confidence: float     # 摘要质量置信度 0-1


class FileParserService:
    """统一文件解析服务，封装 parse_document / ocr_document，加页数限制和 LLM 摘要"""

    @staticmethod
    def _should_parse(file_path: str) -> bool:
        """判断是否需要解析"""
        path = Path(file_path)
        if not path.exists():
            return False
        if path.stat().st_size > MAX_FILE_SIZE_MB * 1024 * 1024:
            logger.info(f"跳过大文件: {path.stat().st_size / 1024 / 1024:.1f}MB > {MAX_FILE_SIZE_MB}MB")
            return False
        return path.suffix.lower() in {'.pdf', '.docx', '.xlsx', '.jpg', '.jpeg', '.png', '.bmp', '.tiff'}

    @staticmethod
    def _choose_tool(file_path: str) -> str:
        """根据扩展名选择解析工具"""
        suffix = Path(file_path).suffix.lower()
        if suffix in {'.jpg', '.jpeg', '.png', '.bmp', '.tiff'}:
            return 'ocr_document'
        return 'parse_document'

    @staticmethod
    async def parse_and_summarize(file_path: str, filename: str) -> ParseResult | None:
        """
        解析文件并生成摘要。
        返回 None 表示跳过或失败。
        """
        if not FileParserService._should_parse(file_path):
            return None

        tool_name = FileParserService._choose_tool(file_path)

        # 延迟导入，避免启动时循环依赖
        from emily_core.tools.registry import BusinessFlowToolRegistry

        try:
            tool = BusinessFlowToolRegistry.get(tool_name)
            if tool is None:
                logger.warning(f"工具未注册: {tool_name}")
                return None

            raw_result = await tool.handler({
                "file_path": file_path,
                "max_pages": MAX_PAGES if tool_name == 'parse_document' else None,
            })

            # 调用 LLM 生成结构化摘要
            summary = await FileParserService._llm_summarize(raw_result, filename)

            return ParseResult(
                summary=summary.get("summary", ""),
                file_type=summary.get("file_type", "其他"),
                key_info=summary.get("key_info", {}),
                confidence=summary.get("confidence", 0.5),
            )

        except Exception as e:
            logger.warning(f"文件解析失败 {filename}: {e}")
            return None

    @staticmethod
    async def _llm_summarize(raw_content: str, filename: str) -> dict:
```

### 步骤 3.1 [执行] 新建定时任务文件

新建文件 `emily-core/emily_core/services/scheduler/jobs/daily_file_parse_job.py`：

```python
import asyncio
import logging

logger = logging.getLogger(__name__)

BATCH_LIMIT = 50
INTER_FILE_DELAY_SEC = 0.5  # 限流，防打爆 LLM


async def daily_file_parse_job():
    """每日批量解析 content_summary 为空的文件"""
    from emily_core.services.file_service import FileService
    from emily_core.services.file_parser_service import FileParserService

    # 延迟导入 M4 的匹配器（M3 执行时 M4 可能尚未部署，用 try/except 兼容）
    try:
        from emily_core.services.node_deliverable_matcher import NodeDeliverableMatcher
        matcher = NodeDeliverableMatcher()
    except ImportError:
        matcher = None

    file_service = FileService()
    pending = await file_service.get_by_summary_null(limit=BATCH_LIMIT)
    logger.info(f"批量解析：待处理 {len(pending)} 个文件")

    for f in pending:
        try:
            result = await FileParserService.parse_and_summarize(
                f.storage_path, f.filename
            )
            if result is None:
                continue

            await file_service.update_summary(str(f.id), result.summary)

            # 触发节点成果匹配（M4 部署后生效）
            if matcher:
                await matcher.match_and_notify(f, result)

        except Exception as e:
            logger.warning(f"批量解析失败 {f.file_no}: {e}")
            continue

        await asyncio.sleep(INTER_FILE_DELAY_SEC)

    logger.info(f"批量解析完成，处理了 {len(pending)} 个文件")
```

## M4：节点成果匹配器

### 步骤 4.1 [执行] 新建 NodeDeliverableMatcher

新建文件 `emily-core/emily_core/services/node_deliverable_matcher.py`：

```python
import difflib
import logging
from dataclasses import dataclass, field

logger = logging.getLogger(__name__)

HIGH_CONFIDENCE = 0.85
MEDIUM_CONFIDENCE = 0.6


@dataclass
class MatchCandidate:
    node_id: str
    node_name: str
    confidence: float
    match_reason: str


class NodeDeliverableMatcher:
    """
    节点成果匹配器。
    输入：(文件对象, ParseResult)
    输出：匹配候选列表，按置信度降序。
    """

    async def match(self, file_obj, parse_result) -> list[MatchCandidate]:
        """
        :param file_obj: files 表的 ORM 对象（有 filename, created_at 等字段）
        :param parse_result: ParseResult（有 file_type, summary 字段）
        :return: 匹配候选列表
        """
        candidates = []

        # 优先级 1：文件名关键词匹配（权重 0.9-1.0）
        filename = getattr(file_obj, 'filename', '')
        keyword_matches = await self._match_by_filename(filename)
        candidates.extend(keyword_matches)

        # 优先级 2：文件类型 + 阶段推断（权重 0.7-0.85）
        if parse_result:
            type_matches = await self._match_by_file_type(parse_result.file_type)
            for m in type_matches:
                # 如果已有同名候选，取最高置信度；否则追加
                existing = [c for c in candidates if c.node_id == m.node_id]
                if not existing:
                    candidates.append(m)

        # 优先级 3：摘要关键词匹配（权重 0.6-0.75）
        if parse_result and parse_result.summary:
            summary_matches = await self._match_by_summary(parse_result.summary)
            for m in summary_matches:
                existing = [c for c in candidates if c.node_id == m.node_id]
                if not existing:
                    candidates.append(m)

        # 优先级 4：时间窗口加分（+0.1）
        candidates = await self._apply_time_bonus(candidates, file_obj)

        # 按置信度降序
        candidates.sort(key=lambda x: x.confidence, reverse=True)
        return candidates

    async def match_and_notify(self, file_obj, parse_result) -> None:
        """匹配并通知管理员（高置信度才通知）"""
        candidates = await self.match(file_obj, parse_result)
        if not candidates:
            return

        best = candidates[0]
        if best.confidence >= HIGH_CONFIDENCE:
            await self._notify_admin(file_obj, best, candidates)
        elif best.confidence >= MEDIUM_CONFIDENCE:
            logger.info(
                f"中置信度匹配: 文件 {file_obj.filename} -> "
                f"节点 {best.node_name} (置信度 {best.confidence:.0%})，不主动通知"
            )

    # ── 各匹配策略具体实现 ──

    async def _match_by_filename(self, filename: str) -> list[MatchCandidate]:
        """用文件名匹配节点成果关键词。AI 执行者实现：从全景节点表加载关键词列表做相似度匹配。"""
        return []

    async def _match_by_file_type(self, file_type: str) -> list[MatchCandidate]:
        """用文件分类（证照/合同/图纸/...）匹配对应阶段的节点。"""
        return []

    async def _match_by_summary(self, summary: str) -> list[MatchCandidate]:
        """用摘要内容关键词匹配节点成果描述。"""
        return []

    async def _apply_time_bonus(
        self, candidates: list[MatchCandidate], file_obj
    ) -> list[MatchCandidate]:
        """如果文件上传时间在节点计划完成窗口内，置信度 +0.1"""
        return candidates

    async def _notify_admin(
        self, file_obj, best: MatchCandidate, all_candidates: list[MatchCandidate]
    ) -> None:
        """向管理员推送匹配通知（含操作选项）。AI 执行者实现：复用 node_tool.py 的通知通道。"""
        pass
```

### 步骤 4.3 [执行] 完善匹配策略的具体实现

上面的 `NodeDeliverableMatcher` 只搭了骨架，每个 `_match_by_*` 方法需要 AI 执行者根据实际项目环境填充：

- **`_match_by_filename`**：从全景节点表加载每个节点的 `deliverable_keywords`，用 `difflib.SequenceMatcher` 计算文件名与关键词的相似度，相似度 > 0.7 的生成候选
- **`_match_by_file_type`**：建立 file_type → 阶段的映射表（如 `证照 → 规划/施工/销售 阶段`），加载该阶段内的所有未完成节点作为候选
- **`_match_by_summary`**：将摘要切词，与节点成果描述做关键词交集，交集数 / 总词数 作为置信度
- **`_apply_time_bonus`**：文件 `created_at` 在节点 `planned_completion_date ± 7天` 内 → +0.1
- **`_notify_admin`**：参考 `node_tool.py` 的通知实现，发送带操作按钮的消息

|--------|-------|------|
| `FILE_PARSE_MAX_PAGES` | 10 | PDF 最多解析页数 |
| `FILE_PARSE_MAX_SIZE_MB` | 50 | 超过此大小的文件跳过 |
| `FILE_PARSE_BATCH_LIMIT` | 50 | 每日批量解析数量上限 |
| `NODE_MATCH_HIGH_CONFIDENCE` | 0.85 | 高置信度阈值（主动通知） |
| `NODE_MATCH_MEDIUM_CONFIDENCE` | 0.6 | 中置信度阈值（记录日志） |

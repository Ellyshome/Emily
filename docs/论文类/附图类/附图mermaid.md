# 图1 传统项目团队信息流与AI信息协同模式对比
```mermaid
flowchart LR

    %% =========================
    %% 传统模式
    %% =========================

    subgraph T["传统项目团队信息协同模式"]
        direction TB

        T_site["项目现场状态"]
        T_contractor["承建单位<br/>（建筑 / 景观 / 精装等分包）"]
        T_dept["项目部<br/>（工程 / 成本 / 设计条线）"]
        T_owner["集团管理层"]

        T_site -->|"信息产生源头"| T_contractor
        T_contractor -->|"上行：现场信息上报"| T_dept
        T_dept -->|"上行：摘要汇总 / 筛选过滤 / 汇报"| T_owner

        T_owner -->|"下行：决策解读 / 任务分工"| T_dept
        T_dept -->|"下行：指令传达 / 要求落地"| T_contractor

        T_owner -.->|"定期巡检"| T_site
        T_dept -.->|"定时巡视"| T_site

        T_issue["信息逐级过滤<br/>过程信息流失<br/>跨部门壁垒<br/>知识随人员流动"]
        T_dept -.-> T_issue
    end


    %% =========================
    %% AI协同模式
    %% =========================

    subgraph A["信息协同型AI团队模式"]
        direction TB

        AI["团队公共AI信息协同中枢<br/>Emily"]

        K["团队公共认知环境<br/>项目态势 + 组织规则 + 系统能力"]

        A_owner["建设单位（集团）<br/>集团管理层"]
        A_contractor["承建单位<br/>（建筑 / 景观 / 精装等分包）"]
        A_dept["项目部<br/>（工程 / 成本 / 设计条线）"]

        A_owner -->|"管理信息 / 决策指令"| AI
        A_contractor -->|"现场信息 / 施工数据"| AI
        A_dept -->|"业务过程信息"| AI

        AI <--> K

        AI -->|"跨源信息 / 过程追溯"| A_owner
        AI -->|"按权限提供信息支持"| A_contractor
        AI -->|"按权限提供信息支持"| A_dept
    end


    %% =========================
    %% 对比关系
    %% =========================

    T -->|"信息协同模式重构"| A


    %% =========================
    %% 问题与效果
    %% =========================

    CMP_T["信息逐级过滤<br/>过程信息流失<br/>跨部门壁垒<br/>知识随人员流动"]
    T --> CMP_T

    CMP_A["持续采集<br/>多源关联<br/>公共认知<br/>按权限服务<br/>全过程可追溯"]
    A --> CMP_A
```
